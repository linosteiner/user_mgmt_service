# === Deployment-Reihenfolge ===

# 1. Traefik zuerst (LoadBalancer-Provisioning dauert am laengsten)
kubectl apply -f k8s/traefik/
kubectl get svc traefik -w          # auf EXTERNAL-IP warten

# 2. DNS: A-Record vcs.linosteiner.ch auf die EXTERNAL-IP umstellen
nslookup vcs.linosteiner.ch

# 3. Restliche Workloads
kubectl apply -f k8s/postgres/
kubectl wait --for=condition=ready pod -l app=postgres --timeout=120s
kubectl apply -f k8s/user-mgmt/ -f k8s/auth-portal/
kubectl apply -f k8s/ingress/

# === Status / Debugging ===
kubectl get pods,svc,ingress
kubectl logs deployment/traefik | grep -i acme     # Zertifikatsausstellung
kubectl logs deployment/postgres
kubectl logs deployment/user-mgmt-service
kubectl logs deployment/auth-portal

# Bei CrashLoopBackOff: Logs des VORHERIGEN Containers
kubectl logs -l app=postgres --previous --tail=100

# Node-Auslastung (der Node ist klein: 920m CPU / 1.5Gi allocatable)
kubectl describe node | sed -n '/Allocated resources/,/Events/p'

# === Verifikation ===
curl -I http://vcs.linosteiner.ch                  # erwartet 301 -> https
curl -I https://vcs.linosteiner.ch                 # erwartet 200
curl -sk https://vcs.linosteiner.ch/api/users      # Backend antwortet (401 = ok, kein 404)

# Traefik-Dashboard (nur lokal, nicht oeffentlich exponiert)
kubectl port-forward svc/traefik-dashboard 8080:8080
# -> http://localhost:8080/dashboard/

# === Let's Encrypt: Staging -> Produktion ===
# In k8s/traefik/traefik-config.yaml die caServer-Zeile entfernen, dann:
kubectl apply -f k8s/traefik/
kubectl delete pod -l app=traefik

# === Aufraeumen ===
kubectl delete -f k8s/ingress/ -f k8s/auth-portal/ -f k8s/user-mgmt/ -f k8s/postgres/ -f k8s/traefik/
